<script type="module">
  import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
  import {
    getDatabase,
    onValue,
    ref,
    runTransaction
  } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-database.js";

  const firebaseConfig = {
    apiKey: "AIzaSyAKDczyyUwdxTn1QP1jkQks8SL91PwZawQ",
    authDomain: "contador-web-gyr.firebaseapp.com",
    databaseURL: "https://contador-web-gyr-default-rtdb.firebaseio.com",
    projectId: "contador-web-gyr"
  };

  const app = initializeApp(firebaseConfig);
  const db = getDatabase(app);
  const visitasRef = ref(db, "visitas");

  runTransaction(visitasRef, (actual) => (actual || 0) + 1);
  onValue(visitasRef, (snapshot) => {
    document.getElementById("visitas").textContent = snapshot.val() ?? 0;
  });
</script>
